x:have no useful
o:other
op:opcode(op1:3 op2:2)
[fct]or[f]:function code(1([f])/2/3/5)
[rgs]:select of registers(3/5)
[rgs_1]&[rgs_2]:select of registers(3)
[r1]&[r2]&[r3]:select of registers(2) (select-x = {0:[rx]})
[2]:select of registers(1)  (select-x = {01:[x]})
[1]select of registers(1)  (select = {10:[1]}
[flg]:ALU Flag enable(make interruption)[h:V l:C] / ALU flag select[11:Z 10:V 01:S 00:C]
[Immediate]:Immediate
[i]:interruption enable
[s]:shamt
[cm]:number of compare(0 or 1)
